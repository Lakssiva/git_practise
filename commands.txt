// version check
git --version

// configuring user name and email
git config --global user.name "Laks"
git config --global user.email "kslakshmipriya13@gmail.com"

// file -> repo (for tracking using git)
git init

// staging ( like ready to be staged)
git add commands.txt
git add .
git add file1 file2 file3

// commiting ( for saving permanently like a snapshot and commit -> a checkpoint )
git commit -m "Created a txt file"

// add and commit to tracked files
git commit -am "added new lines"

// checking
git status

// commit history
git log
git log --oneline

// changes
git diff

// discards unstaged changes and go back to last committed work 
git restore commands.txt

// removes staged file from staging but doesnt delete the content
git restore --staged commands.txt

// --------------github----------------

// check if local and remote are connected
git remote -v

// connect remote
git remote add origin https://github.com/Lakssiva/git_practise.git

// send commits to remote
git push -u origin main
git push

// remote -> local
git pull

// fetch changes or downloads data from remote but it will not be shown in local 
git fetch
git fetch origin branch-name
git fetch --all

// see changes from fetch
git log HEAD..origin/main
git log -p HEAD..origin/main
git diff HEAD origin/main

// fetch and pull
fetch = download updates, don't merge
pull  = fetch + integrate

// ---------branch-----------

// create branch
git branch sample

// switch to branch
git switch sample
git checkout sample

// create and switch
git switch -c sample
git checkout -b sample

// see branches
git branch

// delete a branch
git branch -d login

// check branches in remote
git branch -r

// merge branch to master
git merge sample

// reset (deletes and rewrites history)
git reset --soft commit-id  // uncommited code safe in stage
git reset --hard commit-id

// revert (adds new commit and preserves history)
git revert commit-id

// detached HEAD
head refers to a commit-id and not a branch so new commits are independent and doesnt belong to any branch

// creates detached head so if u switch branches commits will be lost
git checkout commit-id 

// select a specific commit from another branch and copy it to current branch
git cherry-pick commit-id

// keeps unfinished code (no commit has done)
git stash

// restores changes from stash
git stash pop

// rebase (keeps history straight with no merge commits it takes the current branch and moves to target branch and then reapplies the current on top of target branch)
git rebase

// GitHub Actions
automated CI/CD platform built into GitHub that works using triggers like push, pull..

//--------summary---------
I write code → Git notices changes → I stage selected changes → I commit them into history → I push those commits to GitHub → teammates can pull them → we work with branches → create PR → review → merge into main.
