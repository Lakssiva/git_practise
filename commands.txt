// version check
git --version

// configuring user name and email
git config --global user.name "Laks"
git config --global user.email "kslakshmipriya13@gmail.com"

// file -> repo (for tracking using git)
git init

// staging ( like ready to be staged)
git add commands.txt
git add .
git add file1 file2 file3

// commiting ( for saving permanently like a snapshot and commit -> a checkpoint )
git commit -m "Created a txt file"

// checking
git status

// commit history
git log
git log --oneline

// changes
git diff

// discards unstaged changes and go back to last committed work 
git restore commands.txt

// removes staged file from staging but doesnt delete the content
git restore --staged commands.txt

// --------------github----------------

// check if local and remote are connected
git remote -v

// connect remote
git remote add origin https://github.com/Lakssiva/git_practise.git

// send commits to remote
git push -u origin main
git push
